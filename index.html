function getTargetSpreadsheet() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  if (!ss) {
    var file = SpreadsheetApp.getActive();
    if (file) ss = file;
  }
  return ss;
}

// スプレッドシートを開いた時に専用メニューを上部に追加
function onOpen() {
  var ui = SpreadsheetApp.getUi();
  ui.createMenu('受注管理メニュー')
    .addItem('マスタシートを今すぐ作成/初期化', 'initAllSheetsManual')
    .addToUi();
}

function initAllSheetsManual() {
  var ss = getTargetSpreadsheet();
  if (!ss) {
    SpreadsheetApp.getUi().alert('スプレッドシートを取得できませんでした。');
    return;
  }
  initRequiredSheets(ss);
  SpreadsheetApp.getUi().alert('「受注管理」「企業店舗マスタ」「担当者マスタ」の作成・初期化が完了しました！');
}

function initRequiredSheets(ss) {
  if (!ss) {
    ss = getTargetSpreadsheet();
  }
  if (!ss) {
    throw new Error("スプレッドシートが見つかりません。スプレッドシートの「拡張機能」>「Apps Script」から実行されているか確認してください。");
  }

  // ① 受注管理シート
  var orderSheet = ss.getSheetByName("受注管理");
  if (!orderSheet) {
    orderSheet = ss.insertSheet("受注管理");
    orderSheet.appendRow([
      "管理ID", "委託元", "依頼元", "実施日", "ステータス", 
      "エリア", "企業名", "店舗名", "種別", "演者名/取材名", 
      "担当者", "請求金額", "支払金額", "粗利益", "交通費", "特記事項", "同期日時"
    ]);
    orderSheet.getRange(1, 1, 1, 17).setBackground("#f1f5f9").setFontWeight("bold");
  }

  // ② 担当者マスタシート（存在しなければ即座に自動生成）
  var staffSheet = ss.getSheetByName("担当者マスタ");
  if (!staffSheet) {
    staffSheet = ss.insertSheet("担当者マスタ");
    staffSheet.appendRow(["担当者名", "更新日時"]);
    staffSheet.getRange(1, 1, 1, 2).setBackground("#ede9fe").setFontWeight("bold");
    var defaultStaff = ["佐藤", "田中", "高橋", "渡辺", "伊藤"];
    var nowStr = Utilities.formatDate(new Date(), "Asia/Tokyo", "yyyy/MM/dd HH:mm:ss");
    var sRows = defaultStaff.map(function(s) { return [s, nowStr]; });
    staffSheet.getRange(2, 1, sRows.length, 2).setValues(sRows);
  }

  // ③ 企業店舗マスタシート
  var masterSheet = ss.getSheetByName("企業店舗マスタ");
  if (!masterSheet) {
    masterSheet = ss.insertSheet("企業店舗マスタ");
    masterSheet.appendRow(["エリア", "企業名", "店舗名", "更新日時"]);
    masterSheet.getRange(1, 1, 1, 4).setBackground("#e2e8f0").setFontWeight("bold");
  }
}

function doPost(e) {
  var lock = LockService.getScriptLock();
  try {
    lock.waitLock(30000);
  } catch (tErr) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: "Lock timeout" }))
      .setMimeType(ContentService.MimeType.JSON);
  }

  try {
    var ss = getTargetSpreadsheet();
    initRequiredSheets(ss); // 全シートの存在を自動確認・作成

    var payload = JSON.parse(e.postData.contents);
    
    // ① 企業・店舗マスタの同期処理
    if (payload.action === 'sync_master') {
      var mSheet = ss.getSheetByName("企業店舗マスタ");
      mSheet.clear();
      mSheet.appendRow(["エリア", "企業名", "店舗名", "更新日時"]);
      mSheet.getRange(1, 1, 1, 4).setBackground("#e2e8f0").setFontWeight("bold");
      
      var now = Utilities.formatDate(new Date(), "Asia/Tokyo", "yyyy/MM/dd HH:mm:ss");
      var mData = payload.masterData || {};
      var rows = [];
      
      var areaOrder = ["福岡", "佐賀", "長崎", "熊本", "大分", "宮崎", "鹿児島", "沖縄"];
      var allAreas = Object.keys(mData).sort(function(a, b) {
        var idxA = areaOrder.indexOf(a);
        var idxB = areaOrder.indexOf(b);
        if (idxA === -1 && idxB === -1) return a.localeCompare(b, 'ja');
        if (idxA === -1) return 1;
        if (idxB === -1) return -1;
        return idxA - idxB;
      });
      
      allAreas.forEach(function(area) {
        var compObj = mData[area] || {};
        var comps = Object.keys(compObj).sort(function(a, b) { return a.localeCompare(b, 'ja'); });
        comps.forEach(function(comp) {
          var stores = compObj[comp] || [];
          if (stores.length === 0) {
            rows.push([area, comp, "(店舗なし)", now]);
          } else {
            stores.forEach(function(st) {
              rows.push([area, comp, st, now]);
            });
          }
        });
      });
      
      if (rows.length > 0) {
        mSheet.getRange(2, 1, rows.length, 4).setValues(rows);
      }
      return ContentService.createTextOutput(JSON.stringify({ status: "success", count: rows.length }))
        .setMimeType(ContentService.MimeType.JSON);
    }

    // ② 担当者マスタのシート作成・同期処理
    if (payload.action === 'sync_staff' || payload.action === 'create_staff_sheet') {
      var staffSheet = ss.getSheetByName("担当者マスタ");
      staffSheet.clear();
      staffSheet.appendRow(["担当者名", "更新日時"]);
      staffSheet.getRange(1, 1, 1, 2).setBackground("#ede9fe").setFontWeight("bold");
      
      var nowStaff = Utilities.formatDate(new Date(), "Asia/Tokyo", "yyyy/MM/dd HH:mm:ss");
      var sList = payload.staffList || ["佐藤", "田中", "高橋", "渡辺", "伊藤"];
      var sRows = [];
      sList.forEach(function(st) {
        if (st && st.trim() !== '') {
          sRows.push([st.trim(), nowStaff]);
        }
      });
      if (sRows.length > 0) {
        staffSheet.getRange(2, 1, sRows.length, 2).setValues(sRows);
      }
      return ContentService.createTextOutput(JSON.stringify({ status: "success", count: sRows.length }))
        .setMimeType(ContentService.MimeType.JSON);
    }
    
    // ③ 受注データの同期処理
    var orderSheet = ss.getSheetByName("受注管理");
    
    // 全件クリア処理
    if (payload.action === 'clear_all_orders') {
      if (orderSheet.getLastRow() > 1) {
        orderSheet.getRange(2, 1, orderSheet.getLastRow() - 1, 17).clearContent();
      }
      return ContentService.createTextOutput(JSON.stringify({ status: "success", count: 0 }))
        .setMimeType(ContentService.MimeType.JSON);
    }
    
    var data = payload.orders || payload;
    var items = Array.isArray(data) ? data : [data];
    var timestamp = Utilities.formatDate(new Date(), "Asia/Tokyo", "yyyy/MM/dd HH:mm:ss");
    
    // 一括同期（全件送信）の場合は既存データをヘッダー以外クリアして完全再書き込み
    if (payload.action === 'sync_all_orders' || Array.isArray(payload.orders)) {
      if (orderSheet.getLastRow() > 1) {
        orderSheet.getRange(2, 1, orderSheet.getLastRow() - 1, 17).clearContent();
      }
    }
    
    items.forEach(function(item) {
      var billing = Number(item.billing) || 0;
      var payment = Number(item.payment) || 0;
      var profit = billing - payment;
      var statusJa = item.status === 'confirmed' ? '確定' : (item.status === 'tentative' ? '仮案件' : 'バラシ');
      
      orderSheet.appendRow([
        item.id || "",
        item.consignor || item.clientSource || "",
        item.client || "",
        item.date || "",
        statusJa,
        item.area || "",
        item.company || "",
        item.store || "",
        item.type || "",
        item.performer || "",
        item.manager || "",
        billing,
        payment,
        profit,
        item.transportation || "込み",
        item.notes || "",
        timestamp
      ]);
    });
    
    // 追記後に「実施日順(昇順) → 都道府県番号順」で自動ソート
    var lastRow = orderSheet.getLastRow();
    if (lastRow > 2) {
      var prefOrderList = ["福岡", "佐賀", "長崎", "熊本", "大分", "宮崎", "鹿児島", "沖縄"];
      var allRows = orderSheet.getRange(2, 1, lastRow - 1, 17).getValues();
      allRows.sort(function(a, b) {
        var dateA = Utilities.formatDate(new Date(a[3]), "Asia/Tokyo", "yyyy-MM-dd");
        var dateB = Utilities.formatDate(new Date(b[3]), "Asia/Tokyo", "yyyy-MM-dd");
        if (dateA !== dateB) return dateA.localeCompare(dateB);
        var cleanA = (a[5] || "").toString().replace(/[都道府県]/g, "").trim();
        var cleanB = (b[5] || "").toString().replace(/[都道府県]/g, "").trim();
        var idxA = prefOrderList.indexOf(cleanA);
        var idxB = prefOrderList.indexOf(cleanB);
        if (idxA === -1) idxA = 999;
        if (idxB === -1) idxB = 999;
        if (idxA !== idxB) return idxA - idxB;
        return (a[5] || "").toString().localeCompare((b[5] || "").toString(), "ja");
      });
      orderSheet.getRange(2, 1, allRows.length, 17).setValues(allRows);
    }
    
    return ContentService.createTextOutput(JSON.stringify({ status: "success", count: items.length }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}

function doGet(e) {
  try {
    var ss = getTargetSpreadsheet();
    initRequiredSheets(ss); // 全シートの存在を自動確認・作成
    
    var action = (e && e.parameter && e.parameter.action) ? e.parameter.action : 'get_orders';

    // 接続維持用のping確認
    if (action === 'ping') {
      return ContentService.createTextOutput(JSON.stringify({ status: "success", time: Date.now() }))
        .setMimeType(ContentService.MimeType.JSON);
    }

    // ④ 受注案件一覧の取得
    if (action === 'get_orders') {
      var oSheet = ss.getSheetByName("受注管理");
      if (!oSheet || oSheet.getLastRow() <= 1) {
        return ContentService.createTextOutput(JSON.stringify({ status: "success", data: [] }))
          .setMimeType(ContentService.MimeType.JSON);
      }
      var rows = oSheet.getRange(2, 1, oSheet.getLastRow() - 1, 17).getValues();
      var orderList = [];
      rows.forEach(function(r) {
        if (!r[3] && !r[7]) return;
        var statusStr = r[4] ? r[4].toString() : '';
        var statusKey = 'confirmed';
        if (statusStr === '仮案件') statusKey = 'tentative';
        else if (statusStr === 'バラシ') statusKey = 'cancelled';

        var formattedDate = "";
        if (r[3] instanceof Date) {
          formattedDate = Utilities.formatDate(r[3], "Asia/Tokyo", "yyyy-MM-dd");
        } else if (r[3]) {
          formattedDate = r[3].toString().replace(/\//g, '-');
        }

        orderList.push({
          id: r[0] ? r[0].toString() : 'ORD-' + Date.now(),
          consignor: r[1] ? r[1].toString() : '',
          client: r[2] ? r[2].toString() : '',
          date: formattedDate,
          status: statusKey,
          area: r[5] ? r[5].toString() : '',
          company: r[6] ? r[6].toString() : '',
          store: r[7] ? r[7].toString() : '',
          type: r[8] ? r[8].toString() : '',
          performer: r[9] ? r[9].toString() : '',
          manager: r[10] ? r[10].toString() : '',
          billing: Number(r[11]) || 0,
          payment: Number(r[12]) || 0,
          transportation: r[14] ? r[14].toString() : '込み',
          notes: r[15] ? r[15].toString() : ''
        });
      });
      return ContentService.createTextOutput(JSON.stringify({ status: "success", data: orderList }))
        .setMimeType(ContentService.MimeType.JSON);
    }

    // 担当者マスタの取得
    if (action === 'get_staff') {
      var sSheet = ss.getSheetByName("担当者マスタ");
      if (!sSheet || sSheet.getLastRow() <= 1) {
        return ContentService.createTextOutput(JSON.stringify({ status: "success", data: [] }))
          .setMimeType(ContentService.MimeType.JSON);
      }
      var sRows = sSheet.getRange(2, 1, sSheet.getLastRow() - 1, 1).getValues();
      var staffRes = [];
      sRows.forEach(function(r) {
        if (r[0] && r[0].toString().trim() !== '' && staffRes.indexOf(r[0].toString().trim()) === -1) {
          staffRes.push(r[0].toString().trim());
        }
      });
      return ContentService.createTextOutput(JSON.stringify({ status: "success", data: staffRes }))
        .setMimeType(ContentService.MimeType.JSON);
    }

    // 企業店舗マスタの取得
    var mSheet = ss.getSheetByName("企業店舗マスタ");
    if (!mSheet || mSheet.getLastRow() <= 1) {
      return ContentService.createTextOutput(JSON.stringify({ status: "success", data: {} }))
        .setMimeType(ContentService.MimeType.JSON);
    }
    var rows = mSheet.getRange(2, 1, mSheet.getLastRow() - 1, 3).getValues();
    var result = {};
    rows.forEach(function(row) {
      var area = row[0];
      var comp = row[1];
      var store = row[2];
      if (!area || !comp) return;
      if (!result[area]) result[area] = {};
      if (!result[area][comp]) result[area][comp] = [];
      if (store && store !== "(店舗なし)" && result[area][comp].indexOf(store) === -1) {
        result[area][comp].push(store);
      }
    });
    return ContentService.createTextOutput(JSON.stringify({ status: "success", data: result }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
